cmake_minimum_required(VERSION 3.20)
project(slam_mer_posterior_tests LANGUAGES CXX)
find_package(GTSAM REQUIRED)
find_package(Threads REQUIRED)
# Match the production build: GTSAM's colocated cephes dependency needs
# transitive RPATH when validating against an installed GTSAM package.
if(UNIX AND NOT APPLE)
  add_link_options(-Wl,--disable-new-dtags)
endif()
add_executable(test_gaussian_posterior test_gaussian_posterior.cpp ../src/gaussian_posterior.cpp ../src/sim3_factor_derivatives.cpp)
target_include_directories(test_gaussian_posterior PRIVATE ../include)
target_link_libraries(test_gaussian_posterior PRIVATE gtsam Threads::Threads)
target_compile_features(test_gaussian_posterior PRIVATE cxx_std_17)
enable_testing()
foreach(group linear pose edge publication)
  add_test(NAME posterior_${group} COMMAND test_gaussian_posterior ${group})
endforeach()
add_executable(benchmark_gaussian_posterior benchmark_gaussian_posterior.cpp ../src/gaussian_posterior.cpp ../src/sim3_factor_derivatives.cpp)
target_include_directories(benchmark_gaussian_posterior PRIVATE ../include)
target_link_libraries(benchmark_gaussian_posterior PRIVATE gtsam Threads::Threads)
target_compile_features(benchmark_gaussian_posterior PRIVATE cxx_std_17)
